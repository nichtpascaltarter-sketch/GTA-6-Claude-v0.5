int leashHand(int ped, u32 pedUid) {
    using namespace wild_detail;
    if (ped < 0) return -1;
    for (const Group& G : gW.groups)
        if (G.used && G.type == GT_DOG_LEASH && G.owner == ped && (pedUid == 0 || G.ownerUid == pedUid)) return G.leashHand;
    return -1;
}
